cmake_minimum_required(VERSION 3.20)
project(GetUsageProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Require Windows x64 MSVC SDK toolchain")
endif()
add_executable(dxgi_getusage_probe probe.cpp)
target_compile_features(dxgi_getusage_probe PRIVATE cxx_std_17)
target_compile_definitions(dxgi_getusage_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(dxgi_getusage_probe PRIVATE /EHsc /utf-8)
set_property(TARGET dxgi_getusage_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(dxgi_getusage_probe PRIVATE d3d11 dxgi)
