cmake_minimum_required(VERSION 3.20)
project(DxmtResourceUsageSemantics LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Require the existing Windows x64 SDK toolchain")
endif()
add_library(dxmt_resource_usage_semantics OBJECT semantics.cpp)
target_compile_features(dxmt_resource_usage_semantics PRIVATE cxx_std_17)
target_compile_definitions(dxmt_resource_usage_semantics PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(dxmt_resource_usage_semantics PRIVATE /EHsc)
