cmake_minimum_required(VERSION 3.20)
project(CfxMainVendorTests LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang"
   OR NOT CMAKE_C_COMPILER_TARGET STREQUAL "x86_64-pc-windows-msvc")
  message(FATAL_ERROR "Use the original Windows x64 clang-cl/MSVC-SDK toolchain")
endif()
get_filename_component(CFX_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CFX_MAIN_BUILTINS_ARCHIVE "" CACHE FILEPATH "Verified byte-identical MAIN /MT archive snapshot")
if(NOT EXISTS "${CFX_MAIN_BUILTINS_ARCHIVE}")
  message(FATAL_ERROR "Run native/windows-client-tests/run.py build first")
endif()
add_library(main_compiler_rt STATIC IMPORTED)
set_target_properties(main_compiler_rt PROPERTIES IMPORTED_LOCATION "${CFX_MAIN_BUILTINS_ARCHIVE}")
add_library(main_compiler_rt_fixture SHARED "${CFX_PROJECT_ROOT}/native/compiler-rt/fixture.c")
target_link_libraries(main_compiler_rt_fixture PRIVATE main_compiler_rt)
target_link_options(main_compiler_rt_fixture PRIVATE "/MAP:${CMAKE_BINARY_DIR}/main-compiler-rt-fixture.map")
set_target_properties(main_compiler_rt_fixture PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(main_compiler_rt_probe "${CFX_PROJECT_ROOT}/native/compiler-rt/probe.c")
target_include_directories(main_compiler_rt_probe PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(main_compiler_rt_probe PRIVATE main_compiler_rt_fixture)
set_target_properties(main_compiler_rt_probe PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)

option(CFX_MAIN_OPENSSL_TESTS "Build own known-answer consumer of the existing MAIN crypto archive" OFF)
if(CFX_MAIN_OPENSSL_TESTS)
  set(CFX_MAIN_CRYPTO_ARCHIVE "" CACHE FILEPATH "Verified byte-identical MAIN OpenSSL /MT archive snapshot")
  if(NOT EXISTS "${CFX_MAIN_CRYPTO_ARCHIVE}")
    message(FATAL_ERROR "Run native/windows-client-tests/openssl.py build first")
  endif()
  add_library(main_openssl_crypto STATIC IMPORTED)
  set_target_properties(main_openssl_crypto PROPERTIES IMPORTED_LOCATION "${CFX_MAIN_CRYPTO_ARCHIVE}")
  add_executable(main_openssl_probe openssl_probe.c)
  target_include_directories(main_openssl_probe PRIVATE "${CMAKE_BINARY_DIR}"
    "${CFX_PROJECT_ROOT}/port/fivem/code/vendor/openssl"
    "${CFX_PROJECT_ROOT}/port/fivem/vendor/openssl/include")
  target_link_libraries(main_openssl_probe PRIVATE main_openssl_crypto main_compiler_rt ws2_32 crypt32 advapi32 bcrypt)
  target_link_options(main_openssl_probe PRIVATE "/MAP:${CMAKE_BINARY_DIR}/main-openssl-probe.map")
  set_target_properties(main_openssl_probe PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
